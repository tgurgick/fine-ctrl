#### Service Interfaces (NEW)
```python
# backend/services/interfaces.py
from abc import ABC, abstractmethod
from typing import List, Optional
from uuid import UUID

class IAgentService(ABC):
    @abstractmethod
    async def analyze_task(self, description: str) -> dict:
        """Analyze task and return configuration"""
        pass

    @abstractmethod
    async def generate_examples(self, task_id: UUID, count: int) -> List[dict]:
        """Generate training examples"""
        pass

class ITrainingService(ABC):
    @abstractmethod
    async def create_job(self, task_id: UUID, dataset_id: UUID) -> dict:
        """Create training job"""
        pass

    @abstractmethod
    async def get_status(self, job_id: UUID) -> dict:
        """Get job status"""
        pass

class IEvaluationService(ABC):
    @abstractmethod
    async def evaluate_model(self, model_id: UUID) -> dict:
        """Evaluate model performance"""
        pass

class IDeploymentService(ABC):
    @abstractmethod
    async def deploy_model(self, model_id: UUID) -> dict:
        """Deploy model to inference endpoint"""
        pass
```

```python
from fastapi import APIRouter, Depends
from backend.api.deps import get_current_user, require_task_ownership
from backend.services.interfaces import IAgentService
from backend.schemas.task import TaskCreate, TaskResponse

router = APIRouter()

@router.post("/api/tasks", response_model=TaskResponse)
@limiter.limit("10/minute")
async def create_task(
    request: TaskCreate,
    current_user: User = Depends(get_current_user),
    agent_service: IAgentService = Depends(get_agent_service),
    db: Session = Depends(get_db)
):
    # Sanitize input
    safe_description = validate_and_sanitize(request.description)

    # Call agent service (will be mock for now)
    analysis = await agent_service.analyze_task(safe_description)

    # Create task
    task = Task(
        id=uuid4(),
        user_id=current_user.id,
        name=request.name,
        description=safe_description,
        config=analysis
    )

    db.add(task)
    await db.commit()

    return task

@router.get("/api/tasks/{task_id}", response_model=TaskResponse)
async def get_task(
    task: Task = Depends(require_task_ownership)  # Enforces authorization
):
    return task
```

```python
# backend/services/deployment.py
class DeploymentService:
    async def deploy_model(
        self,
        model_version_id: UUID,
        user_id: UUID,
        name: str,
        is_public: bool = False
    ) -> Deployment:
        """Deploy model to Modal - simplified for MVP"""

        # Create deployment record
        deployment = Deployment(
            id=uuid4(),
            model_version_id=model_version_id,
            user_id=user_id,
            name=name,
            is_public=is_public,
            status="deploying"
        )

        db.add(deployment)
        await db.commit()

        # Deploy to Modal
        endpoint_url = await self._deploy_to_modal(model_version_id, deployment.id)

        # Update with URL
        deployment.endpoint_url = endpoint_url
        deployment.status = "active"
        await db.commit()

        return deployment

# deployment/inference.py
@app.function(gpu="T4", keep_warm=1)
@modal.web_endpoint(method="POST")
async def inference(request: dict):
    """Simple inference endpoint - MVP version"""

    # Extract deployment ID from URL path or header
    deployment_id = request.get("deployment_id")

    # Check if public or user has access (simple check)
    deployment = await get_deployment(deployment_id)

    if not deployment.is_public:
        # Require authentication
        user_id = await validate_user_token(request.headers.get("Authorization"))
        if user_id != deployment.user_id:
            raise HTTPException(403, "Access denied")

    # Run inference
    result = await model.generate(request["input"])

    # Log usage
    await log_inference(deployment_id)

    return {"output": result}
```
